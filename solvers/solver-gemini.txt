import numpy as np
from itertools import product

# Define the base module as a set of sub-voxel coordinates (6x6x6 space for target cube)
# Base module spans 2x2x4 in sub-voxels (or 1x1x2 in unit measure)
# Rectangular part: 2x2x2, Triangular wedge part: slope along one axis.

def get_base_module():
    voxels = set()
    # 2x2x2 rectangular base
    for x, y, z in product(range(2), range(2), range(2)):
        voxels.add((x, y, z))
    # Triangular wedge extending along x/z
    for x in range(2):
        for y in range(2):
            for z in range(2, 4 - x): # slanted face
                voxels.add((x, y, z))
    return voxels

# Define 24 standard 3D rotations for a shape
def get_rotations(voxel_set):
    rotations = []
    coords = np.array(list(voxel_set))
    
    # Generate all 24 rotational matrices
    for x_rot in range(4):
        for y_rot in range(4):
            for z_rot in range(4):
                # Standard rotation matrices
                R_x = np.array([[1,0,0],[0,0,-1],[0,1,0]])
                R_y = np.array([[0,0,1],[0,1,0],[-1,0,0]])
                R_z = np.array([[0,-1,0],[1,0,0],[0,0,1]])
                
                R = np.linalg.matrix_power(R_x, x_rot) @ \
                    np.linalg.matrix_power(R_y, y_rot) @ \
                    np.linalg.matrix_power(R_z, z_rot)
                
                rotated = np.dot(coords, R.T)
                # Normalize to origin (min coord = 0)
                rotated -= rotated.min(axis=0)
                # Convert back to tuple of sorted tuples for uniqueness
                rot_tuple = tuple(sorted(map(tuple, rotated)))
                if rot_tuple not in rotations:
                    rotations.append(rot_tuple)
    return rotations

# Backtracking solver to place 9 pieces in a 6x6x6 sub-voxel grid
def solve_cube(pieces, grid_size=(6, 6, 6)):
    grid = np.zeros(grid_size, dtype=int)
    solution = {}

    def can_place(shape, pos):
        for vx, vy, vz in shape:
            x, y, z = pos[0] + vx, pos[1] + vy, pos[2] + vz
            if not (0 <= x < grid_size[0] and 0 <= y < grid_size[1] and 0 <= z < grid_size[2]):
                return False
            if grid[x, y, z] != 0:
                return False
        return True

    def place(shape, pos, piece_id):
        for vx, vy, vz in shape:
            grid[pos[0] + vx, pos[1] + vy, pos[2] + vz] = piece_id

    def remove(shape, pos):
        for vx, vy, vz in shape:
            grid[pos[0] + vx, pos[1] + vy, pos[2] + vz] = 0

    def backtrack(piece_idx):
        if piece_idx > len(pieces):
            return True
        
        piece_rotations = pieces[piece_idx]
        
        # Find first empty voxel to optimize search tree
        empty_voxels = np.argwhere(grid == 0)
        if len(empty_voxels) == 0:
            return piece_idx > len(pieces)
        
        target_pos = tuple(empty_voxels[0])

        for rot in piece_rotations:
            # Shift rotation so it covers the target empty voxel
            for ref_voxel in rot:
                pos = (target_pos[0] - ref_voxel[0], 
                       target_pos[1] - ref_voxel[1], 
                       target_pos[2] - ref_voxel[2])
                
                if can_place(rot, pos):
                    place(rot, pos, piece_idx)
                    solution[piece_idx] = (pos, rot)
                    if backtrack(piece_idx + 1):
                        return True
                    remove(rot, pos)
                    del solution[piece_idx]
        return False

    if backtrack(1):
        print("Solution found!")
        return solution
    else:
        print("No solution exists.")
        return None
